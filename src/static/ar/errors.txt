# رسائل أخطاء MikroTik Hotspot — العربية
# MikroTik Hotspot Portal 2026

# أخطاء داخلية
internal-error = حدث خطأ تقني مؤقت أثناء الاتصال، يرجى المحاولة مرة أخرى.
config-error = تعذر الاتصال بسبب إعدادات الشبكة، يرجى التواصل مع الدعم الفني.

# أخطاء المصادقة
not-logged-in = يرجى تسجيل الدخول أولاً لفتح هذه الصفحة.
invalid-username = كود الكارت أو كلمة المرور غير صحيحة؛ تأكد من البيانات وأعد المحاولة.
wrong-mac-username = هذا الكارت مرتبط بجهاز آخر ولا يمكن استخدامه من هذا الهاتف.
invalid-mac = هذا الكارت غير مسموح باستخدامه من هذا الجهاز، يرجى التواصل مع الدعم الفني.
chap-missing = المتصفح لا يدعم تسجيل الدخول الآمن؛ يرجى تفعيل JavaScript أو الفتح في متصفح Chrome.

# حدود الجلسة
user-session-limit = هذا الكارت مستخدم حالياً على الحد الأقصى من الأجهزة؛ افصل الجهاز الآخر أولاً.
license-session-limit = تم الوصول للحد الأقصى من المتصلين حالياً، يرجى الانتظار والمحاولة بعد قليل.

# حدود الحساب
uptime-limit = انتهى الوقت المتاح لهذه البطاقة. يرجى استخدام كارت جديد.
traffic-limit = انتهى الرصيد (سعة التحميل) المتاح. يرجى استخدام كارت جديد.
account-expired = انتهت مدة صلاحية الكارت. يرجى استخدام كارت جديد.

# أخطاء الشبكة
ippool-empty = الشبكة مزدحمة حالياً ولا توجد مساحات اتصال متاحة، انتظر قليلاً وأعد المحاولة.
shutting-down = خدمة الإنترنت متوقفة مؤقتاً للصيانة، يرجى المحاولة بعد قليل.

# أخطاء RADIUS
radius-timeout = تعذر التحقق من الكارت حالياً بسبب بطء الخادم؛ انتظر لحظات وأعد المحاولة.
auth-in-progress = جارٍ التحقق من الكارت بالفعل، يرجى الانتظار لحظات.
# النص الخام يُمرّر ليترجمه القالب مباشرة عبر config.json (radiusErrorTranslations)
radius-reply = $(error-orig)

# عام
simultaneous-session-limit = لديك اتصال نشط بالفعل على جهاز آخر؛ قم بتسجيل الخروج منه أولاً.
